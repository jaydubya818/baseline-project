   ```typescript
   interface BuyerProfile {
     // Existing fields
     capitalAvailableMin: number;
     capitalAvailableMax: number;
     yearsOfExperience: number;
     industryExperience: string[];
     profileCompleteness: number;
     canAccessNDA: boolean; // New field to track NDA access eligibility
   }
   ```

2. Create a middleware to check profile completeness before NDA access:
   ```typescript
   const ndaProfileGateMiddleware = async (req, res, next) => {
     const userId = req.user.id;
     const buyerProfile = await prisma.buyerProfile.findUnique({
       where: { userId }
     });
     
     if (!buyerProfile) {
       return res.status(403).json({ 
         error: 'Profile required', 
         message: 'You must create a buyer profile before accessing confidential information' 
       });
     }
     
     // Check required fields
     const hasRequiredFields = 
       buyerProfile.capitalAvailableMin !== null &&
       buyerProfile.capitalAvailableMax !== null &&
       buyerProfile.yearsOfExperience !== null &&
       buyerProfile.industryExperience?.length > 0;
       
     if (!hasRequiredFields) {
       return res.status(403).json({ 
         error: 'Incomplete profile', 
         message: 'Please complete your buyer profile with capital range and experience information' 
       });
     }
     
     // Update canAccessNDA flag if needed
     if (!buyerProfile.canAccessNDA) {
       await prisma.buyerProfile.update({
         where: { userId },
         data: { canAccessNDA: true }
       });
     }
     
     next();
   };
   ```

5. Create a profile completeness calculator function that factors in NDA requirements:
   ```typescript
   function calculateProfileCompleteness(profile: BuyerProfile): number {
     const requiredFields = [
       'firstName', 'lastName', 'email', 'phone',
       'capitalAvailableMin', 'capitalAvailableMax',
       'yearsOfExperience', 'industryExperience'
     ];
     
     const completedFields = requiredFields.filter(field => 
       profile[field] !== null && profile[field] !== undefined && 
       (Array.isArray(profile[field]) ? profile[field].length > 0 : true)
     );
     
     return Math.round((completedFields.length / requiredFields.length) * 100);
   }
   ```
