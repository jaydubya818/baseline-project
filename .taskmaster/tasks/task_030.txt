4. Implement an embedding generation service:
   ```typescript
   import { OpenAI } from 'openai';
   import { PrismaClient } from '@prisma/client';

   const openai = new OpenAI({
     apiKey: process.env.OPENAI_API_KEY,
   });
   const prisma = new PrismaClient();

   export async function generateEmbedding(text: string): Promise<number[]> {
     const response = await openai.embeddings.create({
       model: "text-embedding-ada-002",
       input: text,
     });
     return response.data[0].embedding;
   }

   export async function updateListingEmbedding(listingId: string): Promise<void> {
     const listing = await prisma.listing.findUnique({
       where: { id: listingId },
     });
     
     if (!listing) throw new Error('Listing not found');
     
     // Combine relevant text fields for embedding
     const textToEmbed = `${listing.title} ${listing.description} ${listing.assetType}`;
     const embedding = await generateEmbedding(textToEmbed);
     
     await prisma.listing.update({
       where: { id: listingId },
       data: {
         contentEmbedding: embedding,
       },
     });
   }
   ```

5. Create a semantic search service:
   ```typescript
   export async function semanticSearch(query: string, limit: number = 10): Promise<any[]> {
     const queryEmbedding = await generateEmbedding(query);
     
     // Raw SQL query for vector similarity search
     const results = await prisma.$queryRaw`
       SELECT id, title, description, 
         1 - (contentEmbedding <=> ${queryEmbedding}::vector) as similarity
       FROM "Listing"
       WHERE contentEmbedding IS NOT NULL
       ORDER BY similarity DESC
       LIMIT ${limit};
     `;
     
     return results;
   }
   ```

6. Implement API endpoints for semantic search:
   ```typescript
   // routes/search.ts
   import { Router } from 'express';
   import { semanticSearch } from '../services/vectorService';

   const router = Router();

   router.get('/semantic', async (req, res) => {
     try {
       const { query, limit = 10 } = req.query;
       if (!query || typeof query !== 'string') {
         return res.status(400).json({ error: 'Query parameter is required' });
       }
       
       const results = await semanticSearch(query, Number(limit));
       return res.json(results);
     } catch (error) {
       console.error('Semantic search error:', error);
       return res.status(500).json({ error: 'Failed to perform semantic search' });
     }
   });

   export default router;
   ```

8. Implement frontend components for semantic search:
   ```tsx
   // components/SemanticSearchBar.tsx
   import { useState } from 'react';
   import axios from 'axios';

   export default function SemanticSearchBar() {
     const [query, setQuery] = useState('');
     const [results, setResults] = useState([]);
     const [loading, setLoading] = useState(false);
     
     const handleSearch = async () => {
       if (!query.trim()) return;
       
       setLoading(true);
       try {
         const response = await axios.get('/api/search/semantic', {
           params: { query }
         });
         setResults(response.data);
       } catch (error) {
         console.error('Search failed:', error);
       } finally {
         setLoading(false);
       }
     };
     
     return (
       <div>
         <div className="search-container">
           <input
             type="text"
             value={query}
             onChange={(e) => setQuery(e.target.value)}
             placeholder="Search for similar listings..."
           />
           <button onClick={handleSearch} disabled={loading}>
             {loading ? 'Searching...' : 'Search'}
           </button>
         </div>
         
         {results.length > 0 && (
           <div className="results-container">
             {results.map((item) => (
               <div key={item.id} className="result-item">
                 <h3>{item.title}</h3>
                 <p>{item.description}</p>
                 <span>Similarity: {(item.similarity * 100).toFixed(2)}%</span>
               </div>
             ))}
           </div>
         )}
       </div>
     );
   }
   ```
