   ```prisma
   model AdvisorProfile {
     id             String   @id @default(uuid())
     userId         String   @unique
     user           User     @relation(fields: [userId], references: [id])
     specialties    String[]
     experience     Int      // Years of experience
     bio            String   @db.Text
     availability   String   // JSON string of availability schedule
     maxDealLoad    Int      @default(5)
     currentDeals   Int      @default(0)
     responseTime   Int      // Target response time in minutes
     managedDeals   ManagedDeal[]
     createdAt      DateTime @default(now())
     updatedAt      DateTime @updatedAt
   }
   
   model ManagedDeal {
     id             String   @id @default(uuid())
     dealroomId     String   @unique
     dealroom       Dealroom @relation(fields: [dealroomId], references: [id])
     advisorId      String
     advisor        AdvisorProfile @relation(fields: [advisorId], references: [id])
     serviceTier    String   // "SELF_SERVICE", "GUIDED", "MANAGED"
     status         String   // "ACTIVE", "PAUSED", "COMPLETED"
     startDate      DateTime @default(now())
     endDate        DateTime?
     notes          String?  @db.Text
     nextActions    Json?    // Array of next action items
     lastActivity   DateTime @default(now())
     createdAt      DateTime @default(now())
     updatedAt      DateTime @updatedAt
   }
   ```
