```typescript
// Add to DealWorkflowService
async checkAndAdvanceStage(workflowId: string): Promise<boolean> {
  const currentStage = await this.getCurrentStage(workflowId);
  const isComplete = await this.checkStageCompletion(currentStage.id);
  
  if (isComplete) {
    await this.advanceStage(workflowId);
    return true;
  }
  
  return false;
}
```

10. Create smart routing system:
```typescript
class SmartRoutingService {
  // Get next appropriate page/action based on workflow state
  async getNextRoute(workflowId: string, userId: string): Promise<string> {
    const workflow = await workflowService.getWorkflow(workflowId);
    const currentStage = workflow.stages.find(s => s.stageNumber === workflow.currentStage);
    const userActions = await workflowService.getNextActionsForUser(workflowId, userId);
    
    if (userActions.length > 0) {
      const priorityAction = userActions[0];
      return this.getRouteForAction(priorityAction);
    }
    
    return `/dealrooms/${workflow.dealroomId}`;
  }
  
  private getRouteForAction(action: StageAction): string {
    // Logic to determine appropriate route based on action type
    switch(action.actionType) {
      case 'FORM_SUBMISSION':
        return `/forms/${action.id}`;
      case 'DOCUMENT_UPLOAD':
        return `/documents/upload/${action.id}`;
      // Other action types
      default:
        return `/actions/${action.id}`;
    }
  }
}
```
