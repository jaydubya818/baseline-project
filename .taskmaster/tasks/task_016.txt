## 1. Payment Idempotency

### Stripe Idempotency Keys
- Use Stripe idempotency keys on the `POST /api/listings/:id/upgrade` flow
- Generate a deterministic idempotency key using format: 
  `listing_upgrade_${listingId}_${userId}_${timestamp_bucket}`
  (where timestamp_bucket = floor(Date.now() / 60000) for 1-min windows)
- Pass to Stripe checkout session creation:
  ```typescript
  const session = await stripe.checkout.sessions.create({
    // ... config
  }, {
    idempotencyKey: `listing_upgrade_${listingId}_${userId}_${timestampBucket}`
  })
  ```

### Local Persistence Before Stripe Call
- **Create `ListingUpgrade` model in Prisma schema:**
  ```prisma
  model ListingUpgrade {
    id                      String               @id @default(cuid())
    listingId               String
    userId                  String
    externalChargeId        String?              @unique  // Stripe payment_intent ID
    stripeCheckoutSessionId String?              @unique
    idempotencyKey          String               @unique
    status                  UpgradeStatus        @default(PENDING)
    upgradeType             String               // "premium", "featured"
    amount                  Int                  // Amount in cents
    currency                String               @default("usd")
    expiresAt               DateTime?
    refundedAt              DateTime?
    refundReason            String?
    failureReason           String?
    attempts                Int                  @default(1)
    createdAt               DateTime             @default(now())
    updatedAt               DateTime             @updatedAt
    
    listing                 Listing              @relation(fields: [listingId], references: [id])
    user                    User                 @relation(fields: [userId], references: [id])
    
    @@index([listingId])
    @@index([userId])
    @@index([status])
    @@index([expiresAt])
  }

  enum UpgradeStatus {
    PENDING           // Created locally, waiting for Stripe
    PROCESSING        // Stripe session created, awaiting payment
    SUCCEEDED         // Payment confirmed via webhook
    FAILED            // Payment failed
    REFUNDED          // Full refund issued
    PARTIALLY_REFUNDED
    EXPIRED           // Upgrade period ended
    CANCELLED         // User cancelled before payment
  }
  ```

- **Flow for POST /api/listings/:id/upgrade:**
  1. Generate idempotencyKey
  2. Check for existing pending/processing upgrade with same idempotencyKey
  3. Create ListingUpgrade record with status=PENDING before calling Stripe
  4. Call Stripe to create checkout session
  5. Update ListingUpgrade with stripeCheckoutSessionId, status=PROCESSING
  6. Return session URL to client

### Stripe Webhook Handler (`processListingUpgradeWebhook`)

```typescript
// Add to routeEventToHandler switch statement:
case 'checkout.session.completed':

// In handleCheckoutSessionCompleted, add listing_upgrade handling:

async function handleListingUpgradeCompleted(session: Stripe.Checkout.Session): Promise<void> {
  const { metadata, payment_intent, id: sessionId } = session
  const listingId = metadata?.listingId
  const userId = metadata?.userId
  
  if (!listingId || !userId) {
    logger.error('Missing metadata in listing upgrade session', { sessionId })
    return
  }

  // Idempotency check by externalChargeId (payment_intent)
  const existingUpgrade = await prisma.listingUpgrade.findFirst({
    where: {
      OR: [
        { externalChargeId: payment_intent as string },
        { stripeCheckoutSessionId: sessionId }
      ],
      status: 'SUCCEEDED'
    }
  })

  if (existingUpgrade) {
    logger.info('Listing upgrade already processed', { 
      sessionId, 
      upgradeId: existingUpgrade.id 
    })
    return
  }

  // Atomic transaction to update upgrade and listing
  await prisma.$transaction(async (tx) => {
    // Update or create upgrade record
    await tx.listingUpgrade.upsert({
      where: { stripeCheckoutSessionId: sessionId },
      create: {
        listingId,
        userId,
        stripeCheckoutSessionId: sessionId,
        externalChargeId: payment_intent as string,
        idempotencyKey: `webhook_${sessionId}`,
        status: 'SUCCEEDED',
        upgradeType: metadata.upgradeType || 'premium',
        amount: session.amount_total || 0,
        expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000) // 30 days
      },
      update: {
        externalChargeId: payment_intent as string,
        status: 'SUCCEEDED',
        expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000)
      }
    })

    // Apply benefits to listing
    await tx.listing.update({
      where: { id: listingId },
      data: {
        isPremium: true,
        // Add other benefits based on upgradeType
      }
    })
  })

  logger.info('Listing upgrade completed', { listingId, userId, sessionId })
}
```

### Retry/Reconciliation Job
Create `lib/jobs/upgrade-reconciliation.ts`:
```typescript
/**
 * Reconciliation job for stuck ListingUpgrade records
 * Run via cron every 15 minutes
 */
export async function reconcileStuckUpgrades(): Promise<void> {
  const stuckUpgrades = await prisma.listingUpgrade.findMany({
    where: {
      status: { in: ['PENDING', 'PROCESSING'] },
      createdAt: { lt: new Date(Date.now() - 30 * 60 * 1000) } // 30+ mins old
    }
  })

  for (const upgrade of stuckUpgrades) {
    try {
      if (upgrade.stripeCheckoutSessionId) {
        const session = await stripe.checkout.sessions.retrieve(
          upgrade.stripeCheckoutSessionId
        )
        
        if (session.payment_status === 'paid') {
          // Webhook was lost, process now
          await handleListingUpgradeCompleted(session)
        } else if (session.status === 'expired') {
          await prisma.listingUpgrade.update({
            where: { id: upgrade.id },
            data: { status: 'CANCELLED', failureReason: 'Session expired' }
          })
        }
      }
    } catch (error) {
      logger.error('Reconciliation failed for upgrade', { 
        upgradeId: upgrade.id, 
        error 
      })
    }
  }
}
```

## 3. Refunds & Cancellations

### Refund Policy
- Full refunds available within 24 hours of purchase if benefits unused
- Pro-rated refunds (optional) for early cancellation after 24 hours
- No refunds if listing received premium-tier engagement (inquiries, etc.)

### Refund/Cancel Endpoint
`POST /api/listings/:id/upgrade/refund`:
```typescript
export async function POST(req: NextRequest, { params }: { params: { id: string } }) {
  const user = await getCurrentUser()
  const { reason } = await req.json()

  const upgrade = await prisma.listingUpgrade.findFirst({
    where: {
      listingId: params.id,
      userId: user.id,
      status: 'SUCCEEDED'
    },
    include: { listing: true }
  })

  if (!upgrade) {
    return NextResponse.json({ error: 'No active upgrade found' }, { status: 404 })
  }

  // Check refund eligibility
  const hoursSincePurchase = (Date.now() - upgrade.createdAt.getTime()) / (1000*60*60)
  if (hoursSincePurchase > 24 && upgrade.listing.inquiryCount > 0) {
    return NextResponse.json({ 
      error: 'Refunds not available after 24 hours with engagement' 
    }, { status: 400 })
  }

  // Process refund via Stripe
  const refund = await stripe.refunds.create({
    payment_intent: upgrade.externalChargeId,
    reason: 'requested_by_customer'
  })

  // Update records and revoke benefits
  await prisma.$transaction([
    prisma.listingUpgrade.update({
      where: { id: upgrade.id },
      data: { 
        status: 'REFUNDED', 
        refundedAt: new Date(),
        refundReason: reason 
      }
    }),
    prisma.listing.update({
      where: { id: params.id },
      data: { isPremium: false }
    })
  ])

  return NextResponse.json({ success: true, refundId: refund.id })
}
```

/**
 * Background job to handle upgrade expirations
 * Run via cron daily at midnight UTC
 */
export async function processUpgradeExpirations(): Promise<void> {
  const now = new Date()

  // Find expired upgrades
  const expiredUpgrades = await prisma.listingUpgrade.findMany({
    where: {
      status: 'SUCCEEDED',
      expiresAt: { lte: now }
    },
    include: { listing: true }
  })

  for (const upgrade of expiredUpgrades) {
    await prisma.$transaction([
      // Mark upgrade as expired
      prisma.listingUpgrade.update({
        where: { id: upgrade.id },
        data: { status: 'EXPIRED' }
      }),
      // Revoke benefits from listing
      prisma.listing.update({
        where: { id: upgrade.listingId },
        data: {
          isPremium: false,
          premiumExpiresAt: null
        }
      })
    ])

    // Send expiration notification
    await sendNotification({
      userId: upgrade.userId,
      type: 'UPGRADE_EXPIRED',
      data: { listingId: upgrade.listingId }
    })

    logger.info('Upgrade expired', { 
      listingId: upgrade.listingId, 
      upgradeId: upgrade.id 
    })
  }
}

/**
 * Send renewal reminders 3 days before expiration
 */
export async function sendRenewalReminders(): Promise<void> {
  const threeDaysFromNow = new Date(Date.now() + 3 * 24 * 60 * 60 * 1000)
  const fourDaysFromNow = new Date(Date.now() + 4 * 24 * 60 * 60 * 1000)

  const expiringUpgrades = await prisma.listingUpgrade.findMany({
    where: {
      status: 'SUCCEEDED',
      expiresAt: {
        gte: threeDaysFromNow,
        lt: fourDaysFromNow
      }
    }
  })

  for (const upgrade of expiringUpgrades) {
    await sendNotification({
      userId: upgrade.userId,
      type: 'UPGRADE_EXPIRING_SOON',
      data: { 
        listingId: upgrade.listingId,
        expiresAt: upgrade.expiresAt 
      }
    })
  }
}

## 5. Error Handling Strategies

### Network Retries & Exponential Backoff
```typescript
// lib/stripe-client.ts
async function stripeWithRetry<T>(
  operation: () => Promise<T>,
  maxRetries = 3
): Promise<T> {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await operation()
    } catch (error: any) {
      const isRetryable = 
        error.type === 'StripeConnectionError' ||
        error.type === 'StripeAPIError' ||
        error.statusCode >= 500

      if (!isRetryable || attempt === maxRetries) throw error

      const delay = Math.min(1000 * Math.pow(2, attempt), 10000)
      await new Promise(r => setTimeout(r, delay))
      
      logger.warn('Retrying Stripe operation', { 
        attempt, delay, error: error.message 
      })
    }
  }
  throw new Error('Unexpected retry loop exit')
}
```

### Handling Specific Errors
| Error Type              | Handling Strategy                                          |
|-------------------------|-----------------------------------------------------------|
| Card declined           | Return user-friendly message, suggest alternative payment |
| Rate limit (429)        | Exponential backoff, max 5 retries                        |
| Network timeout         | Retry with idempotency key                                |
| Webhook delivery failure| Stripe auto-retries; reconciliation job catches gaps      |
| DB failure after Stripe | Reconciliation job processes from Stripe state            |
| Invalid payment_intent  | Log error, mark upgrade as FAILED with reason             |
| Duplicate webhook       | Idempotency check via externalChargeId returns early      |
